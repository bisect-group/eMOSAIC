# emulator_bench commands for eMOSAIC-T (Ki). Run from the eMOSAIC repo root with the mldb env:
#   cd baselines/eMOSAIC
# --base_dir defaults to <EMULaToR>/data/processed/datasets/kinetic_dataset/ki.
# Results go to results/<task>/ and the embedding cache to results/embeddings.

# 0) Optional: build the protein (ESMFold) and ligand caches first.
python -u emulator_bench/cache_embeddings.py \
  --gpus 0 1 \
  --max_seq_len 700 \
  --chunk_size 64

# 1) All default splits x seeds 666 777 888 (caches features first if needed).
python -u emulator_bench/run_split_benchmarks.py \
  --seeds 666 777 888 \
  --gpus 0 1 \
  --pin_memory \
  --preload_proteins

# 2) Mean and variance over seeds.
python emulator_bench/aggregate_results.py --results_dir results/ki

# 3) One split, one seed.
python -u emulator_bench/train_single_target_tvt.py \
  --train_path ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/train.parquet \
  --val_path   ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/val.parquet \
  --test_path  ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/test.parquet \
  --out_dir results/ki/random_splits_grouped_sequence/seed_666 \
  --seed 666 --pin_memory --preload_proteins

# 4) Predict from a checkpoint.
python -u emulator_bench/predict_single_target.py \
  --input_path ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/test.parquet \
  --ckpt_path results/ki/random_splits_grouped_sequence/seed_666/bestmodel.pth \
  --out_csv results/ki/random_splits_grouped_sequence/seed_666/predictions_test.csv
