# emulator_bench commands for eMOSAIC-T (Ki). Run from baselines/eMOSAIC (the submodule root).
# --base_dir defaults to <EMULaToR>/data/processed/datasets/kinetic_dataset/ki.
# Runs go to $OUT/ki/<split_group>/<split_name>/seed_<n>/ and the cache to $OUT/embeddings.
PY=/home/adhil/.conda/envs/mldb/bin/python
OUT=../../outputs/baselines/eMOSAIC

# 0) Optional: build the protein (ESMFold) and ligand caches first (the launcher below
#    caches on cuda:0 unless you pass --cache_device or --skip_cache).
#    ESMFold runs with the ESM-2 LM in fp16 and bf16 autocast (add --fp32 for full fp32, which needs >22 GB at 700 residues). Old v2 entries (3,495 of 5,936 Ki proteins)
#    are copied into proteins_v3 first; only the missing proteins are computed.
$PY -u emulator_bench/cache_embeddings.py \
  --gpus 1 \
  --max_seq_len 700 \
  --chunk_size 64

# 1) All default splits x seeds 666 777 888 (caches features first if needed).
$PY -u emulator_bench/run_split_benchmarks.py \
  --seeds 666 777 888 \
  --gpus 0 1 \
  --pin_memory \
  --preload_proteins

# 1b) One bf16 run on GPU 1, seed 666, caches already built (a full ki run needs about 14 GB, so two do not fit on a 22 GB A10).
$PY -u emulator_bench/run_split_benchmarks.py \
  --seeds 666 \
  --gpus 1 \
  --runs_per_gpu 1 \
  --amp \
  --skip_cache

# 1c) Add --no_gpu_protein_table to train_single_target_tvt.py or predict_single_target.py if GPU memory is short.

# 2) Mean and variance over seeds.
$PY emulator_bench/aggregate_results.py --results_dir $OUT/ki

# 3) One split, one seed.
$PY -u emulator_bench/train_single_target_tvt.py \
  --train_path ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/train.parquet \
  --val_path   ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/val.parquet \
  --test_path  ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/test.parquet \
  --out_dir $OUT/ki/random_splits_grouped_sequence/all/seed_666 \
  --embeddings_dir $OUT/embeddings --seed 666 --pin_memory --preload_proteins

# 4) Predict from a checkpoint.
$PY -u emulator_bench/predict_single_target.py \
  --input_path ../../data/processed/datasets/kinetic_dataset/ki/random_splits_grouped_sequence/test.parquet \
  --ckpt_path $OUT/ki/random_splits_grouped_sequence/all/seed_666/bestmodel.pth \
  --embeddings_dir $OUT/embeddings \
  --out_csv $OUT/ki/random_splits_grouped_sequence/all/seed_666/predictions_test.csv
